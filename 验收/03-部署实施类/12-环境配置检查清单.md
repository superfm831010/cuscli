## 1. 概述

### 1.1 文档目的

本文档提供 **Cuscli AI 编程助手平台 v1.1.0** 部署前的环境配置检查清单。通过系统化的检查流程，确保部署环境满足系统运行的所有必要条件，降低部署失败风险，提高部署成功率。

### 1.2 适用对象

- **系统管理员**：负责环境准备和配置
- **运维工程师**：负责系统部署实施
- **技术支持人员**：负责部署问题排查
- **项目经理**：负责部署进度管理

### 1.3 使用说明

本检查清单按照以下步骤使用：

1. **部署前检查**：在开始部署前，逐项完成所有检查项
2. **记录检查结果**：在每个检查项后标注 ✅ 或 ❌
3. **处理不合格项**：对于不合格的检查项，按照建议措施进行整改
4. **复查确认**：整改完成后重新检查，确保所有项目合格
5. **签字确认**：所有检查项通过后，由相关人员签字确认

## 7. 综合检查脚本

### 7.1 Linux 自动检查脚本

创建 `cuscli_env_check.sh`：

```bash
#!/bin/bash
# Cuscli 环境配置自动检查脚本 (Linux)

set -e

# 颜色定义
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
BLUE='\033[0;34m'
NC='\033[0m' # No Color

# 结果统计
TOTAL_CHECKS=0
PASSED_CHECKS=0
FAILED_CHECKS=0

# 检查函数
check_item() {
    local name="$1"
    local command="$2"
    local requirement="$3"

    TOTAL_CHECKS=$((TOTAL_CHECKS + 1))
    echo -n "检查 $name ... "

    if eval "$command" > /dev/null 2>&1; then
        echo -e "${GREEN}✅ 通过${NC} ($requirement)"
        PASSED_CHECKS=$((PASSED_CHECKS + 1))
        return 0
    else
        echo -e "${RED}❌ 失败${NC} ($requirement)"
        FAILED_CHECKS=$((FAILED_CHECKS + 1))
        return 1
    fi
}

echo "=========================================="
echo "  Cuscli 环境配置检查脚本"
echo "  版本: v1.0"
echo "  日期: $(date '+%Y-%m-%d %H:%M:%S')"
echo "=========================================="
echo ""

# 1. 硬件资源检查
echo -e "${BLUE}【1. 硬件资源检查】${NC}"
echo "--------------------"

# CPU核心数
CPU_CORES=$(nproc)
echo "CPU 核心数: $CPU_CORES"
check_item "CPU核心数(≥2)" "[ $CPU_CORES -ge 2 ]" "最低2核"

# 内存大小
TOTAL_MEM=$(free -m | awk 'NR==2{print $2}')
echo "总内存: ${TOTAL_MEM}MB"
check_item "内存大小(≥4GB)" "[ $TOTAL_MEM -ge 4000 ]" "最低4GB"

# 磁盘空间
ROOT_AVAIL=$(df -BG / | awk 'NR==2{print $4}' | sed 's/G//')
echo "根分区可用空间: ${ROOT_AVAIL}GB"
check_item "磁盘空间(≥20GB)" "[ $ROOT_AVAIL -ge 20 ]" "最低20GB"

echo ""

# 2. 操作系统检查
echo -e "${BLUE}【2. 操作系统检查】${NC}"
echo "--------------------"

# OS 类型
if [ -f /etc/os-release ]; then
    source /etc/os-release
    echo "操作系统: $PRETTY_NAME"
fi

# 内核版本
KERNEL=$(uname -r)
echo "内核版本: $KERNEL"

# 系统架构
ARCH=$(uname -m)
echo "系统架构: $ARCH"
check_item "系统架构(x86_64)" "[ '$ARCH' = 'x86_64' ]" "必须x86_64"

echo ""

# 3. Python 环境检查
echo -e "${BLUE}【3. Python 环境检查】${NC}"
echo "--------------------"

# Python 版本
if command -v python3 &> /dev/null; then
    PY_VERSION=$(python3 --version | awk '{print $2}')
    echo "Python 版本: $PY_VERSION"

    # 检查版本范围
    PY_MAJOR=$(python3 -c 'import sys; print(sys.version_info.major)')
    PY_MINOR=$(python3 -c 'import sys; print(sys.version_info.minor)')

    if [ "$PY_MAJOR" -eq 3 ] && [ "$PY_MINOR" -ge 10 ] && [ "$PY_MINOR" -le 12 ]; then
        echo -e "${GREEN}✅ Python版本符合要求 (3.10-3.12)${NC}"
        PASSED_CHECKS=$((PASSED_CHECKS + 1))
    else
        echo -e "${RED}❌ Python版本不符合要求，需要3.10-3.12${NC}"
        FAILED_CHECKS=$((FAILED_CHECKS + 1))
    fi
    TOTAL_CHECKS=$((TOTAL_CHECKS + 1))
else
    echo -e "${RED}❌ 未安装 Python3${NC}"
    FAILED_CHECKS=$((FAILED_CHECKS + 1))
    TOTAL_CHECKS=$((TOTAL_CHECKS + 1))
fi

# pip 检查
check_item "pip" "command -v pip3" "必需"

# venv 检查
check_item "venv模块" "python3 -m venv --help" "必需"

echo ""

# 4. 编译工具检查
echo -e "${BLUE}【4. 编译工具检查】${NC}"
echo "--------------------"

check_item "gcc" "command -v gcc" "必需"
check_item "g++" "command -v g++" "必需"
check_item "make" "command -v make" "必需"

echo ""

# 5. Git 检查
echo -e "${BLUE}【5. Git 检查】${NC}"
echo "--------------------"

if command -v git &> /dev/null; then
    GIT_VERSION=$(git --version | awk '{print $3}')
    echo "Git 版本: $GIT_VERSION"
    check_item "Git(≥2.20)" "git --version" "必需"
else
    echo -e "${RED}❌ 未安装 Git${NC}"
    FAILED_CHECKS=$((FAILED_CHECKS + 1))
    TOTAL_CHECKS=$((TOTAL_CHECKS + 1))
fi

echo ""

# 6. 网络检查
echo -e "${BLUE}【6. 网络环境检查】${NC}"
echo "--------------------"

# DNS 解析
check_item "DNS解析(baidu.com)" "ping -c 1 -W 2 baidu.com" "推荐"

# 外网连接（可选）
check_item "外网连接(pypi.org)" "ping -c 1 -W 2 pypi.org" "可选"

echo ""

# 7. 权限检查
echo -e "${BLUE}【7. 权限检查】${NC}"
echo "--------------------"

# 当前用户
CURRENT_USER=$(whoami)
echo "当前用户: $CURRENT_USER"

# sudo 权限
if sudo -n true 2>/dev/null; then
    echo -e "${GREEN}✅ 有 sudo 权限${NC}"
    PASSED_CHECKS=$((PASSED_CHECKS + 1))
else
    echo -e "${YELLOW}⚠️  无免密 sudo 权限${NC}"
fi
TOTAL_CHECKS=$((TOTAL_CHECKS + 1))

echo ""

# 总结
echo "=========================================="
echo -e "${BLUE}【检查总结】${NC}"
echo "=========================================="
echo "总检查项: $TOTAL_CHECKS"
echo -e "${GREEN}通过: $PASSED_CHECKS${NC}"
echo -e "${RED}失败: $FAILED_CHECKS${NC}"

if [ $FAILED_CHECKS -eq 0 ]; then
    echo ""
    echo -e "${GREEN}✅ 所有检查项通过，环境满足部署要求！${NC}"
    exit 0
else
    echo ""
    echo -e "${RED}❌ 存在 $FAILED_CHECKS 项不合格，请先整改！${NC}"
    exit 1
fi
```

### 7.2 Windows 自动检查脚本

创建 `cuscli_env_check.ps1`：

```powershell
# Cuscli 环境配置自动检查脚本 (Windows PowerShell)

$TotalChecks = 0
$PassedChecks = 0
$FailedChecks = 0

function Check-Item {
    param(
        [string]$Name,
        [scriptblock]$Check,
        [string]$Requirement
    )

    $script:TotalChecks++
    Write-Host -NoNewline "检查 $Name ... "

    try {
        $result = & $Check
        if ($result) {
            Write-Host "✅ 通过" -ForegroundColor Green -NoNewline
            Write-Host " ($Requirement)"
            $script:PassedChecks++
            return $true
        } else {
            Write-Host "❌ 失败" -ForegroundColor Red -NoNewline
            Write-Host " ($Requirement)"
            $script:FailedChecks++
            return $false
        }
    } catch {
        Write-Host "❌ 失败" -ForegroundColor Red -NoNewline
        Write-Host " ($Requirement)"
        $script:FailedChecks++
        return $false
    }
}

Write-Host "==========================================" -ForegroundColor Cyan
Write-Host "  Cuscli 环境配置检查脚本 (Windows)" -ForegroundColor Cyan
Write-Host "  版本: v1.0" -ForegroundColor Cyan
Write-Host "  日期: $(Get-Date -Format 'yyyy-MM-dd HH:mm:ss')" -ForegroundColor Cyan
Write-Host "==========================================" -ForegroundColor Cyan
Write-Host ""

# 1. 硬件资源检查
Write-Host "【1. 硬件资源检查】" -ForegroundColor Blue
Write-Host "--------------------"

# CPU
$cpu = Get-WmiObject -Class Win32_Processor
Write-Host "CPU: $($cpu.Name)"
Write-Host "CPU 核心数: $($cpu.NumberOfCores)"
Check-Item "CPU核心数(≥2)" { $cpu.NumberOfCores -ge 2 } "最低2核"

# 内存
$memory = Get-WmiObject -Class Win32_ComputerSystem
$totalMemGB = [math]::Round($memory.TotalPhysicalMemory / 1GB, 2)
Write-Host "总内存: ${totalMemGB}GB"
Check-Item "内存大小(≥4GB)" { $totalMemGB -ge 4 } "最低4GB"

# 磁盘空间
$disk = Get-PSDrive C
$freeSpaceGB = [math]::Round($disk.Free / 1GB, 2)
Write-Host "C盘可用空间: ${freeSpaceGB}GB"
Check-Item "磁盘空间(≥20GB)" { $freeSpaceGB -ge 20 } "最低20GB"

Write-Host ""

# 2. 操作系统检查
Write-Host "【2. 操作系统检查】" -ForegroundColor Blue
Write-Host "--------------------"

$os = Get-ComputerInfo
Write-Host "操作系统: $($os.WindowsProductName)"
Write-Host "系统版本: $($os.WindowsVersion)"
Write-Host "系统架构: $($os.OsArchitecture)"
Check-Item "系统架构(64-bit)" { $os.OsArchitecture -eq "64-bit" } "必须64位"

Write-Host ""

# 3. Python 环境检查
Write-Host "【3. Python 环境检查】" -ForegroundColor Blue
Write-Host "--------------------"

try {
    $pyVersion = python --version 2>&1
    Write-Host "Python 版本: $pyVersion"

    # 检查版本范围
    $versionMatch = $pyVersion -match "Python 3\.(10|11|12)\."
    if ($versionMatch) {
        Write-Host "✅ Python版本符合要求 (3.10-3.12)" -ForegroundColor Green
        $PassedChecks++
    } else {
        Write-Host "❌ Python版本不符合要求，需要3.10-3.12" -ForegroundColor Red
        $FailedChecks++
    }
    $TotalChecks++
} catch {
    Write-Host "❌ 未安装 Python" -ForegroundColor Red
    $FailedChecks++
    $TotalChecks++
}

# pip 检查
Check-Item "pip" { $null -ne (Get-Command pip -ErrorAction SilentlyContinue) } "必需"

Write-Host ""

# 4. Git 检查
Write-Host "【4. Git 检查】" -ForegroundColor Blue
Write-Host "--------------------"

try {
    $gitVersion = git --version 2>&1
    Write-Host "Git 版本: $gitVersion"
    Check-Item "Git" { $null -ne (Get-Command git -ErrorAction SilentlyContinue) } "必需"
} catch {
    Write-Host "❌ 未安装 Git" -ForegroundColor Red
    $FailedChecks++
    $TotalChecks++
}

Write-Host ""

# 5. 网络检查
Write-Host "【5. 网络环境检查】" -ForegroundColor Blue
Write-Host "--------------------"

Check-Item "网络连接(baidu.com)" { Test-Connection -ComputerName baidu.com -Count 1 -Quiet } "推荐"

Write-Host ""

# 总结
Write-Host "==========================================" -ForegroundColor Cyan
Write-Host "【检查总结】" -ForegroundColor Blue
Write-Host "==========================================" -ForegroundColor Cyan
Write-Host "总检查项: $TotalChecks"
Write-Host "通过: $PassedChecks" -ForegroundColor Green
Write-Host "失败: $FailedChecks" -ForegroundColor Red

if ($FailedChecks -eq 0) {
    Write-Host ""
    Write-Host "✅ 所有检查项通过，环境满足部署要求！" -ForegroundColor Green
    exit 0
} else {
    Write-Host ""
    Write-Host "❌ 存在 $FailedChecks 项不合格，请先整改！" -ForegroundColor Red
    exit 1
}
```
